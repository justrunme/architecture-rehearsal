🌐 ACCESSING THE API
--------------------
The control plane API is available at:

  Service: {{ include "architecture-rehearsal.fullname" . }}
  Port: {{ .Values.service.port }}
{{- if eq .Values.service.type "ClusterIP" }}

  # From within the cluster:
  export REHEARSAL_API_URL="http://{{ include "architecture-rehearsal.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}"
  export REHEARSAL_API_TOKEN="<your-token>"

  # Port forward for local access:
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "architecture-rehearsal.fullname" . }} 8080:{{ .Values.service.port }}
  export REHEARSAL_API_URL="http://localhost:8080"
{{- else if eq .Values.service.type "NodePort" }}

  # Get the NodePort:
  export NODE_PORT=$(kubectl get -n {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "architecture-rehearsal.fullname" . }})
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "API available at http://$NODE_IP:$NODE_PORT"
{{- else if eq .Values.service.type "LoadBalancer" }}

  # Wait for LoadBalancer IP:
  kubectl get svc -n {{ .Release.Namespace }} {{ include "architecture-rehearsal.fullname" . }} -w

  # Once assigned:
  export SERVICE_IP=$(kubectl get svc -n {{ .Release.Namespace }} {{ include "architecture-rehearsal.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  export REHEARSAL_API_URL="http://$SERVICE_IP:{{ .Values.service.port }}"
{{- end }}

  # Test the API:
  curl -H "Authorization: Bearer $REHEARSAL_API_TOKEN" $REHEARSAL_API_URL/v1/version
{{- end }}
